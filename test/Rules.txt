Random rand = new Random();

Events.UserMessageCallback UserMessageCallback = null;

public void OnStartup(){

	if (SHOW_CHAT_ADS){
		CreateTimer(CHAT_ADS_INTERVAL, 0, "Chat_DisplayAd", "");
		if (CHAT_ADS_DISPLAY_ONE_ON_START) Chat_DisplayAd(new TriggerArgs(null, null, false));
	}

	if (SHOW_POPUP_AD){
		Game.ShowPopupMessage(POPUP_AD_CONTENT, POPUP_TEXT_COLOR);
		if (POPUP_AD_TIME != 0) CreateTimer(POPUP_AD_TIME, 1, "Popup_HideAd", "");
	}

UserMessageCallback = Events.UserMessageCallback.Start(OnUserMessage);
}

public void OnUserMessage(UserMessageCallbackArgs ua) 
{ 
	switch (ua.Command)
{
	case "RULES": 
	Game.ShowPopupMessage(POPUP_AD_CONTENT, POPUP_TEXT_COLOR);
		if (POPUP_AD_TIME != 0) CreateTimer(POPUP_AD_TIME, 1, "Popup_HideAd", "");

	break;
}
}

private int m_curMessageNum = -1;
public void Chat_DisplayAd(TriggerArgs args){
	Game.ShowChatMessage(SERVER_NAME + ": ");
	for(int i = 0; i < CHAT_ADS_CONTENT.Length; i++){
		Game.ShowChatMessage("/MSG" + CHAT_ADS_CONTENT[i]);
	}
}

public void Popup_HideAd(TriggerArgs args){
	Game.HidePopupMessage();
}

private void CreateTimer(int interval, int count, string method, string id) {
	IObjectTimerTrigger timerTrigger = (IObjectTimerTrigger)Game.CreateObject("TimerTrigger"); 
	timerTrigger.SetIntervalTime(interval);
	timerTrigger.SetRepeatCount(count);
	timerTrigger.SetScriptMethod(method);
	timerTrigger.CustomId = id;
	timerTrigger.Trigger();
}
