public void OnStartup()
{
	//pl = (IObject)Game.CreateObject("InvisibleBlock", new Vector2(9999999, 99999999));

	aw.SetRepeatCount(0);
	aw.SetScriptMethod("TickMe");

	if(Convert.ToBoolean(Game.LocalStorage.GetItem("animrp")))Inputer = Events.PlayerKeyInputCallback.Start(Input);
	foreach(IPlayer p in Game.GetPlayers())
	if(Convert.ToBoolean(Game.LocalStorage.GetItem("nickrp")))
		{
			p.SetNametagVisible(false);
			p.SetStatusBarsVisible(false);
		}
}

public void Input(IPlayer p, VirtualKeyInfo[] keyEvents) 
{
	for(int i = 0; i < keyEvents.Length; i ++) 
	if(keyEvents[i].Event == VirtualKeyEvent.Pressed) 
		{
			switch(keyEvents[i].Key) 
				{
					case VirtualKey.DRAW_SPECIAL:
					if(p.KeyPressed(VirtualKey.WALKING))
						{
							p.SetInputEnabled(false);
							p.AddCommand(new PlayerCommand(PlayerCommandType.DeathKneelInfinite));
							Events.UpdateCallback.Start((x) => p.SetInputEnabled(true), 1411u, 1);
						}
					break;
					case VirtualKey.DRAW_GRENADE:
					if(p.KeyPressed(VirtualKey.WALKING))
						{
							p.SetInputEnabled(false);
							p.AddCommand(new PlayerCommand(PlayerCommandType.StaggerInfinite));
							Events.UpdateCallback.Start((x) => p.SetInputEnabled(true), 200u, 1);
						}
					break;
					case VirtualKey.DRAW_MELEE:
					if(p.KeyPressed(VirtualKey.WALKING))
						{
							p.SetInputEnabled(false);
							p.AddCommand(new PlayerCommand(PlayerCommandType.Fall));
							Events.UpdateCallback.Start((x) => p.SetInputEnabled(true), 500u, 1);
						}
					break;
				}
		}
}

public static string FirstCharToUpper(string input)
{
    if (String.IsNullOrEmpty(input))
        throw new ArgumentException("ARGH!");
    return input.First().ToString().ToUpper() + String.Join("", input.Skip(1));
}

string rm = "", rs = "";
public void TickMe(TriggerArgs args)
{
	ss--;
	if(ss == -1){mm--;ss = 59;}
	if(ss <= 9)rs = "0";else rs = "";
	if(mm <= 9)rm = "0";else rs = "";
	if(mm == 0 && ss <= 10 && ss >= 1)Game.PlaySound("MenuOK", Vector2.Zero);
	else if(mm == 0 && ss == 0)
		{
			Game.PlaySound("MenuCancel", Vector2.Zero);
			aw.StopTimer();
			Events.UpdateCallback.Start((x) =>
				{
					Game.HidePopupMessage();
					Timer = false;
					y = 0;
					n = 0;
				}, 500u, 1);
		}
	Game.ShowPopupMessage(rm + mm.ToString() +":"+ rs + ss.ToString(), Color.White);
}
