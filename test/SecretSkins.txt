int playerCounter;

public void OnStartup(){
	playerCounter = Game.GetActiveUsers().Length;
	
	try{	
		
		foreach(IUser user in Game.GetActiveUsers()){
			for(int i = 0;i < 8 ;i++){
				if(user.GameSlotIndex == i && !user.IsBot){
					
					IUser UserPly = user;
					IPlayer UsrPly = UserPly.GetPlayer();
					IProfile UsrPr = UsrPly.GetProfile();
					//Assign Profile
					IProfile AssignedProfile = AssignProfilePlayer(UserPly, UsrPly, UsrPr, i);
					UsrPly.SetProfile(AssignedProfile);
				
				}
			}
		}
	}catch{Game.RunCommand("/MSG Unexpected Error :(");}

	Events.UserMessageCallback.Start(comm);
	Events.UpdateCallback.Start(checkLeave, 1000);
}


////////////////////////////////////////////////////////////////////
public void comm(UserMessageCallbackArgs args){
	if(args.IsCommand && args.User.GetPlayer() != null && (args.Command == "BEAR" || args.Command == "MECH" || args.Command == "FB" || args.Command == "ME")){
		IUser Uply = (IUser)args.User;
		IPlayer ply = Uply.GetPlayer();
		IProfile pr = new IProfile();
		bool remainHead = false;
		bool remainSkin = false;
		bool remainME = false;
		int indexSlot = Uply.GameSlotIndex + 1;
		
		string skinClr1 = Uply.GetProfile().Skin.Color1;
		string skinClr2 = Uply.GetProfile().Skin.Color2;
		string skin = "Normal";
		string clr1 = "ClothingWhite";
		string clr2 = "ClothingGray";

		if(args.CommandArguments == "me" || args.CommandArguments == "head"){
			pr = Uply.GetProfile();
			remainSkin = true;
			remainHead = true;
			remainME = true;
			if(args.CommandArguments == "head"){
				pr.Head = new IProfileClothingItem("None", Uply.GetProfile().Skin.Color1);
				pr.Accessory = new IProfileClothingItem("None", Uply.GetProfile().Skin.Color1);

				remainHead = false;
			}
		}

		if(args.IsCommand && args.Command == "BEAR"){
			pr.Skin = new IProfileClothingItem("BearSkin", Uply.GetProfile().Skin.Color1);
			ply.SetProfile(pr);
			skin = "BearSkin";
			remainME = false;
		}
		else if(args.IsCommand && args.Command == "FB"){
			if(Uply.GetProfile().Head != null){
				clr2 = Uply.GetProfile().Head.Color1;
				
			}else{
				clr2 = Uply.GetProfile().Skin.Color1;
			}
			
			pr.Skin = new IProfileClothingItem("FrankenbearSkin", Uply.GetProfile().Skin.Color2, clr2);

			ply.SetProfile(pr);

			skin = "FrankenbearSkin";
			remainME = false;
		}
		else if(args.IsCommand && args.Command == "MECH"){
			if(Uply.GetProfile().Head != null){
				clr2 = Uply.GetProfile().Head.Color1;
			}else{
				clr2 = Uply.GetProfile().Skin.Color1;
			}
			if(Uply.GetProfile().ChestOver != null){
				clr1 = Uply.GetProfile().ChestOver.Color1;
			}else{
				clr1 = Uply.GetProfile().Skin.Color1;
			}
			pr.Skin = new IProfileClothingItem("MechSkin", clr1, clr2);
			skin = "MechSkin";
			remainME = false;
			ply.SetProfile(pr);
		}
		else if(args.Command == "ME"){
			pr = Uply.GetProfile();
			skin = "Normal";
			ply.SetProfile(pr);
			remainSkin = true;
			remainHead = true;
			remainME = true;
		}
		StoreData(skin, skinClr1, skinClr2, remainME, clr1, clr2, remainSkin, remainHead, indexSlot);
	}
}

public void StoreData(string skin, string skinClr1, string skinClr2, bool remainME, string clr1, string clr2, bool remainSkin, bool remainHead, int indexSlot){
	Game.LocalStorage.SetItem("plyKey"+indexSlot, skin);
	Game.LocalStorage.SetItem("plyKeyClr1_ply"+indexSlot, skinClr1);
	Game.LocalStorage.SetItem("plyKeyClr2_ply"+indexSlot, skinClr2);
	Game.LocalStorage.SetItem("remainME"+indexSlot, remainME);
	Game.LocalStorage.SetItem("plyKeyColor1Head_ply"+indexSlot, clr1);
	Game.LocalStorage.SetItem("plyKeyColor2Body_ply"+indexSlot, clr2);
	Game.LocalStorage.SetItem("plyKeyRemainSkinply"+indexSlot, remainSkin);
	Game.LocalStorage.SetItem("plyKeyRemainHeadply"+indexSlot, remainHead);
}

public IProfile AssignProfilePlayer(IUser Usr, IPlayer Plyer, IProfile Prfl, int index){
	IProfile newPrfl = Prfl;
	index = index + 1;
	IProfile voidPrfl = new IProfile();

	bool remainLocalME = (bool)Game.LocalStorage.GetItem("remainME"+index);
	string localColor1 = (string)Game.LocalStorage.GetItem("plyKeyColor1Head_ply"+index);
	string localColor2 = (string)Game.LocalStorage.GetItem("plyKeyColor2Body_ply"+index);
	

	
	string localSkin = (string)Game.LocalStorage.GetItem("plyKey"+index);
	bool localMantainHead = (bool)Game.LocalStorage.GetItem("plyKeyRemainHeadply"+index);
	bool localMantainSkin = (bool)Game.LocalStorage.GetItem("plyKeyRemainSkinply"+index);
	
	
	//SkinME
	
	if(remainLocalME){
		Plyer.SetProfile(Prfl);
	}else{
		if(localMantainSkin){
			newPrfl = Prfl;
			if(!localMantainHead){
				newPrfl.Head = new IProfileClothingItem("None", localColor1, localColor2);
				newPrfl.Accessory = new IProfileClothingItem("None", localColor1, localColor2);
			}
		}else{
			newPrfl = voidPrfl;
		}
		newPrfl.Skin = new IProfileClothingItem(localSkin, localColor1, localColor2);
	}

	return newPrfl;
}

int[] playersInGame = {0,0,0,0,0,0,0,0};
public void checkLeave(float elapsed){
	IUser[] newPlayerCounter = Game.GetActiveUsers();
	for(int i = 0;i < 8;i++ ){
		playersInGame[i] = 0;
	}
	foreach(IUser user in newPlayerCounter){
		int ind = user.GameSlotIndex;
		if(ind >= 0){
			playersInGame[ind] = 1;
		}
	}
	for(int indx = 0; indx < 8;indx++){
		if(playersInGame[indx] == 0){
			StoreData("Normal", "ClothingWhite", "ClothingWhite", true, "ClothingWhite", "ClothingWhite", true, true, indx+1);
		}
	}
}
