            public static void StartRandomBasicEvent()
            {
                AllEvents[] validEvents = allEvents.Where((e) =>
                {
                    return !e.isMetaEvent && e.isEnabled;
                }).ToArray();

                for (int i = 0; i < (isComboTime ? 3 : 1); i++)
                {
                    int chosenEvent = rng.Next(validEvents.Length);
                    validEvents[chosenEvent].Announce();
                }

                Events.UpdateCallback.Start((dt) =>
                {
                    StartRandomBasicEvent();
                }, (uint)downtime, 1);
            }
            public static void StartMetaEvent()
            {
                AllEvents[] validMetaEvents = allEvents.Where((e) =>
                {
                    return e.isMetaEvent && e.isEnabled;
                }).ToArray();
                int chosenEvent = rng.Next(validMetaEvents.Length);
                if (rng.Next(2) != 0)
                    validMetaEvents[chosenEvent].AnnounceMeta();

                Events.UpdateCallback.Start((dt) =>
                {
                    StartMetaEvent();
                }, (uint)metadowntime, 1);
            }

            virtual protected void Announce()
            {
                if (!isNameHidden)
                {
                    Game.ShowChatMessage("[CHAOS EFFECT]", Color.Green);
                    string name = GetEffectName();
                    if (delay > 0) name += " [" + delay / 1000 + "s]";
                    Game.ShowChatMessage(name, new Color(0, 155, 255));
                    Game.PlaySound("LogoSlam", Vector2.Zero);
                }
                if (timerCallback == null)
                    Begin();
                else
                {
                    timerCallback.Stop();
                    StartTimer();
                }
            }
            virtual protected void AnnounceMeta()
            {
                Game.ShowChatMessage("[CHAOS META]", Color.Red);
                string name = GetEffectName();
                if (delay > 0) name += " [" + delay / 1000 + "s]";
                Game.ShowChatMessage(name, new Color(255, 0, 255));
                Game.PlaySound("TutorialSuccess2", Vector2.Zero);
                Game.ShowPopupMessage("[CHAOS META]\n" + GetEffectName() + " [" + delay / 1000 + "]\n", new Color(255, 0, 255));
                Begin();
            }
            virtual protected void EndAnnounce()
            {
                if (!isMetaEvent)
                {
                    if (!isNameHidden)
                    {
                        Game.ShowChatMessage("'" + GetEffectName() + "' has ended!", new Color(255, 135, 0));
                    }
                }
                else
                {
                    Game.HidePopupMessage();
                    Game.ShowChatMessage("[META]'" + GetEffectName() + "' has ended!", new Color(180, 0, 255));
                }
            }
            abstract protected void Begin();
            virtual protected void End()
            {

            }

            //clothing method:
            protected IProfileClothingItem GetRandomItem(string[] items)
            {
                // a 20 % chance for nothing
                if (rng.NextDouble() < 0.2f || items.Length == 0)
                {
                    return null;
                }
                string itemName = items[rng.Next(0, items.Length)];
                string colorPalette = Game.GetClothingItemColorPaletteName(itemName);
                ColorPalette cp = Game.GetColorPalette(colorPalette);
                string colorA = "", colorB = "", colorC = "";
                if (cp != null)
                {
                    colorA = (cp.PrimaryColorPackages != null && cp.PrimaryColorPackages.Length > 0 ? cp.PrimaryColorPackages[rng.Next(0, cp.PrimaryColorPackages.Length)] : "");
                    colorB = (cp.SecondaryColorPackages != null && cp.SecondaryColorPackages.Length > 0 ? cp.SecondaryColorPackages[rng.Next(0, cp.SecondaryColorPackages.Length)] : "");
                    colorC = (cp.TertiaryColorPackages != null && cp.TertiaryColorPackages.Length > 0 ? cp.TertiaryColorPackages[rng.Next(0, cp.TertiaryColorPackages.Length)] : "");
                }
                return new IProfileClothingItem(itemName, colorA, colorB, colorC);
            }

            protected override void Begin()
            {
                IPlayer[] Players;
                Players = Game.GetPlayers().Where(x => (x != null && !x.IsDead && !x.IsRemoved)).ToArray();
                //Randomize the player list
                IPlayer[] RandomizedPlayers = Players.OrderBy(x => rng.Next()).ToArray();
                //Save last user in list
                IUser lastUser = RandomizedPlayers[RandomizedPlayers.Length - 1].GetUser();
                IUser tempUser;
                for (int i = 0; i < RandomizedPlayers.Length; i++)

                    //Save current players user to temp
                    tempUser = RandomizedPlayers[i].GetUser();

        class RainbowWorld : AllEvents
        {
            int x = 1;
            Events.UpdateCallback update = null;
            List<string[]> originalColors = new List<string[]>();
            List<IObject> objects = new List<IObject>();
            string[] customColors = {"BgRed","BgOrange", "BgYellow", "BgGreen", "BgCyan", "BgBlue", "BgMagenta"};
            protected override string GetEffectName()
            {
                return "Rainbow World";
            }
            protected override void Begin()
            {
                foreach (IObject bgTiles in Game.GetObjects<IObject>().Where(o => o.Name.StartsWith("Bg") || o.Name.StartsWith("FarBg")).ToArray())
                {
                    objects.Add(bgTiles);
                    originalColors.Add(bgTiles.GetColors());
                    bgTiles.SetColor1(customColors[0]);
                    bgTiles.SetColor2(customColors[0]);
                }
                update = Events.UpdateCallback.Start((dt) =>
                {
                    if (x > 0 && x < customColors.Length)
                    {
                        ColorObjects();
                        x++;
                    }
                    else
                    {
                        x = 0;
                        ColorObjects();
                        x++;
                    }
                }, 2000);
                StartTimer();
            }
            void ColorObjects()
            {
                foreach (IObject bgTiles in Game.GetObjects<IObject>().Where(o => o.Name.StartsWith("Bg") || o.Name.StartsWith("FarBg")))
                {
                bgTiles.SetColor1(customColors[x]);
                bgTiles.SetColor2(customColors[x]);
                }
            }
            protected override void End()
            {
                update.Stop();
                for (int i = 0; i < objects.Count; i++)
                    objects[i].SetColors(originalColors[i]);
                originalColors.Clear();
                objects.Clear();
            }
        }

        class PlayerData
        {
            public Vector2 Position;
            public IPlayer Player;
            public PlayerModifiers Mods;
            public MeleeWeaponItem MeleeWpn;
            public MeleeWeaponItem MakeshiftWpn;
            public RifleWeaponItem RifleWpn;
            public HandgunWeaponItem HandgunWpn;
            public PowerupWeaponItem PowerupWpn;
            public ThrownWeaponItem ThrownWpn;
            public PlayerTeam Team;
            public IProfile Profile;
            public bool Nametag;
            public bool Statusbars;
            public int AmmoPrimary;
            public int AmmoSecondary;
            public int AmmoThrown;
            public PlayerHitEffect HitEffect;
            public bool statusbars;
            public PlayerData(IPlayer p)
            {
                Position = p.GetWorldPosition();
                Player = p;
                Mods = p.GetModifiers();
                MeleeWpn = p.CurrentMeleeWeapon;
                MakeshiftWpn = p.CurrentMeleeMakeshiftWeapon;
                RifleWpn = p.CurrentPrimaryWeapon;
                HandgunWpn = p.CurrentSecondaryWeapon;
                PowerupWpn = p.CurrentPowerupItem;
                ThrownWpn = p.CurrentThrownItem;
                Team = p.GetTeam();
                if (p.IsUser)
                    Profile = p.GetUser().GetProfile();
                else
                    Profile = p.GetProfile();
                Nametag = p.GetNametagVisible();
                Statusbars = p.GetStatusBarsVisible();
                AmmoPrimary = p.CurrentPrimaryWeapon.TotalAmmo;
                AmmoSecondary = p.CurrentSecondaryWeapon.TotalAmmo;
                AmmoThrown = p.CurrentThrownItem.CurrentAmmo;
                HitEffect = p.GetHitEffect();
                statusbars = p.GetStatusBarsVisible();
            }
        }
