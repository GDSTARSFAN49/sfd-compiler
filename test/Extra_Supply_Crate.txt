const float RANDOM_CHANCE = 0.5f;

Random rnd = new Random();

public void OnStartup()
{
	if (!Game.IsFirstUpdate)
	{
		return;
	}
	
	Events.UpdateCallback.Start(SpawnRandomBoxes, 2000, 1);
	
}

public void SpawnRandomBoxes(float ms)
{
	foreach (IPlayer plr in Game.GetPlayers())
	{
      if (!plr.IsDead && rnd.NextDouble() <= RANDOM_CHANCE && plr.GetUser() != null)
		{
			Vector2 pos = plr.GetWorldPosition();
			Game.CreateObject("SupplyCrate00", pos + Vector2.UnitY * 10f);
		}
	}
}
